# requirements.txt
# Python dependencies for the Esperanto Robot laptop side.
#
# Install all at once:
#   pip install -r requirements.txt
#
# Optional packages are marked – the system degrades gracefully without them.

# ── Serial communication (REQUIRED) ───────────────────────────────────────────
pyserial>=3.5

# ── Audio playback (REQUIRED for music/TTS) ───────────────────────────────────
pygame>=2.1.0

# ── Offline text-to-speech (RECOMMENDED) ──────────────────────────────────────
pyttsx3>=2.90

# ── Online text-to-speech / Esperanto TTS (OPTIONAL) ─────────────────────────
# gTTS has basic Esperanto support via Google Translate TTS.
gTTS>=2.3.0

# ── Speech recognition for pronunciation mode (OPTIONAL) ─────────────────────
SpeechRecognition>=3.10.0
# PyAudio is required by SpeechRecognition for microphone input:
#   On Linux:   sudo apt install portaudio19-dev && pip install pyaudio
#   On macOS:   brew install portaudio && pip install pyaudio
#   On Windows: pip install pyaudio  (wheels available on PyPI)
pyaudio>=0.2.13

# ── Webcam face tracking (OPTIONAL but recommended) ───────────────────────────
opencv-python>=4.8.0

# ── AI assistant HTTP requests (REQUIRED if using AI mode) ───────────────────
requests>=2.31.0

# ── Pybricks (hub side only – NOT installed on the laptop) ────────────────────
# The hub runs Pybricks MicroPython. Do NOT install pybricks on the laptop.
# Hub firmware: https://pybricks.com/install/mindstorms-ev3/
